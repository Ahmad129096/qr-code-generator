// Set the color theme before the first paint, so the page never flashes // in the wrong colors. Keeps working without JavaScript afterwards. ;(function () { var theme = 'light' try { var pref = window.localStorage.getItem('qr-theme') if (pref === 'dark') theme = 'dark' else if (pref !== 'light') { theme = window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light' } } catch (e) { theme = 'light' } document.documentElement.setAttribute('data-theme', theme) })()
